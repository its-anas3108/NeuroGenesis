# --- Core scientific stack (required by everything) ---------------------------
numpy>=1.24.0
pandas>=2.0.0
scipy>=1.11.0                # statistical tests, effect sizes, KDE

# --- Deep learning (required for the model, training, XAI) --------------------
# Graph learning is implemented in dense pure PyTorch. torch_geometric is
# deliberately NOT a dependency: every graph here has five nodes and a dense
# 5x5 adjacency, so sparse message-passing machinery would add a heavy,
# version-fragile dependency without benefit, while dense tensors keep the
# attention and edge-gate matrices directly inspectable for the dashboard.
torch>=2.0.0

# --- Machine-learning baselines ----------------------------------------------
# Without scikit-learn the classical baselines (logistic regression, gradient
# boosting, random forest, MLP) are skipped and reported as unavailable.
scikit-learn>=1.3.0

# --- Medical imaging (required for M1-M7: the imaging half of the pipeline) ---
# Without these, the cohort and labels still load and the model/training/XAI/
# statistics/dashboard paths still run on cached artifacts, but no MRI can be
# read, preprocessed, atlas-localized or cropped into ROI patches.
nibabel>=5.0.0               # NIfTI I/O
SimpleITK>=2.3.0             # N4 bias correction, anisotropic diffusion, resampling,
                              # and ADNI DICOM-to-MNI152 registration
nilearn>=0.10.0              # Harvard-Oxford atlas, brain masking, MNI152 template
scikit-image>=0.21.0         # CLAHE, marching-cubes surface area
pydicom>=2.4.0                # ADNI raw-DICOM series ingestion

# --- Visualisation -----------------------------------------------------------
matplotlib>=3.7.0
